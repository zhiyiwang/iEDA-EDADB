cmake_minimum_required(VERSION 3.16)
project(sqlite_pk_fk LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Use -DCMAKE_BUILD_TYPE=Release")
endif()
find_package(SQLite3 REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(pk_fk_benchmark pk_fk_benchmark.cpp)
target_compile_features(pk_fk_benchmark PRIVATE cxx_std_17)
target_include_directories(pk_fk_benchmark PRIVATE "${REPO_ROOT}/src/database/edadb/core/include")
target_compile_definitions(pk_fk_benchmark PRIVATE _EDADB_DEBUG_TRACE_SQL_STMT_=0 EDADB_VERBOSE_TABLE_OP=0)
target_compile_options(pk_fk_benchmark PRIVATE -O3 -Wall -Wextra)
target_link_options(pk_fk_benchmark PRIVATE -O3)
target_link_libraries(pk_fk_benchmark PRIVATE SQLite::SQLite3)
