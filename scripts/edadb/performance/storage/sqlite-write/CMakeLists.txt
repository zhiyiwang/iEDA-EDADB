cmake_minimum_required(VERSION 3.16)
project(sqlite_write LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Use Release")
endif()
find_package(SQLite3 REQUIRED)
add_executable(sqlite_write benchmark.cpp)
target_compile_features(sqlite_write PRIVATE cxx_std_20)
target_compile_options(sqlite_write PRIVATE -O3 -g -Wall -Wextra)
target_link_libraries(sqlite_write PRIVATE SQLite::SQLite3)
add_executable(sqlite_write_profile profile.cpp)
target_compile_features(sqlite_write_profile PRIVATE cxx_std_20)
target_compile_options(sqlite_write_profile PRIVATE -O3 -g -Wall -Wextra)
target_link_libraries(sqlite_write_profile PRIVATE SQLite::SQLite3)
add_executable(sqlite_read_write ../sqlite-vs-text/benchmark.cpp)
target_compile_features(sqlite_read_write PRIVATE cxx_std_20)
target_compile_options(sqlite_read_write PRIVATE -O3 -g -Wall -Wextra)
target_link_libraries(sqlite_read_write PRIVATE SQLite::SQLite3)
add_executable(sqlite_interleaved interleaved.cpp)
target_compile_features(sqlite_interleaved PRIVATE cxx_std_20)
target_compile_options(sqlite_interleaved PRIVATE -O3 -g -Wall -Wextra)
target_link_libraries(sqlite_interleaved PRIVATE SQLite::SQLite3)
