cmake_minimum_required(VERSION 3.16)
project(sqlite_index LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Use -DCMAKE_BUILD_TYPE=Release")
endif()
find_package(SQLite3 REQUIRED)
add_executable(index_benchmark benchmark.cpp)
target_compile_features(index_benchmark PRIVATE cxx_std_20)
target_compile_options(index_benchmark PRIVATE -O3 -march=native -mtune=native -Wall -Wextra)
target_link_libraries(index_benchmark PRIVATE SQLite::SQLite3)
