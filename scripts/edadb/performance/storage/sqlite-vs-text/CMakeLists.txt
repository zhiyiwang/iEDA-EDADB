cmake_minimum_required(VERSION 3.16)
project(sqlite_vs_text LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Use Release")
endif()
find_package(SQLite3 REQUIRED)
add_executable(benchmark benchmark.cpp)
target_compile_features(benchmark PRIVATE cxx_std_20)
target_compile_options(benchmark PRIVATE -O3 -march=native -mtune=native -Wall -Wextra)
target_link_libraries(benchmark PRIVATE SQLite::SQLite3)
add_executable(profile_benchmark benchmark.cpp)
target_compile_features(profile_benchmark PRIVATE cxx_std_20)
target_compile_options(profile_benchmark PRIVATE -O3 -g -march=native -mtune=native -Wall -Wextra)
target_compile_definitions(profile_benchmark PRIVATE SQLITE_TEXT_PERF=1)
target_link_libraries(profile_benchmark PRIVATE SQLite::SQLite3)
add_executable(select_projection select_projection.cpp)
target_compile_features(select_projection PRIVATE cxx_std_20)
target_compile_options(select_projection PRIVATE -O3 -march=native -mtune=native -Wall -Wextra)
target_link_libraries(select_projection PRIVATE SQLite::SQLite3)
